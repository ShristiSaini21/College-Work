#Q20
dd if=/dev/zero of=$HOME/workflow.img bs=1M count=300
WFDEV=$(sudo losetup --find --show --partscan $HOME/workflow.img)
sudo parted -s $WFDEV mklabel msdos mkpart primary 1MiB 250MiB
sudo partprobe $WFDEV
WFPART=${WFDEV}p1
lsblk $WFDEV
sudo mkfs.ext4 $WFPART
sudo mkdir -p /mnt/labfs
sudo mount $WFPART /mnt/labfs
findmnt /mnt/labfs
df -hT /mnt/labfs
echo "Workflow data" | sudo tee /mnt/labfs/data.txt
ls -l /mnt/labfs
cat /mnt/labfs/data.txt
sudo umount /mnt/labfs
WFUUID=$(sudo blkid -s UUID -o value $WFPART)
echo $WFUUID
sudo cp /etc/fstab /etc/fstab.bak
echo "UUID=$WFUUID /mnt/labfs ext4 defaults,nofail 0 2" | sudo tee -a /etc/fstab
sudo findmnt --verify
sudo systemctl daemon-reload
sudo mount -a
findmnt /mnt/labfs
cat /mnt/labfs/data.txt
sudo umount /mnt/labfs
sudo e2fsck -f $WFPART
echo $?
findmnt /mnt/labfs || echo "not mounted"
sudo sed -i '\|/mnt/labfs|d' /etc/fstab
diff /etc/fstab.bak /etc/fstab && echo "fstab restored"
sudo systemctl daemon-reload
sudo mount -a
sudo losetup -a
sudo losetup -d $WFDEV
sudo losetup -d $TESTDEV
rm -f $HOME/labfs.img $HOME/workflow.img
sudo rmdir /mnt/labfs
sudo rm -f /etc/fstab.bak
lsblk
findmnt /mnt/labfs || echo "clean"
grep labfs /etc/fstab || echo "no lab entries in fstab"
