#Q19
#!/bin/bash
if [ $# -lt 1 ]; then
    echo "Usage: $0 <directory>"
    exit 1
fi
dir="$1"
if [ ! -d "$dir" ]; then
    echo "Directory does not exist: $dir"
    exit 1
fi
count=0
for file in "$dir"/*.bak; do
    if [ -f "$file" ]; then
        echo "Backup found: $file"
        count=$((count + 1))
    fi
done
echo "Total backup files: $count"
if [ "$count" -eq 0 ]; then
    echo "$(date '+%Y-%m-%d %H:%M:%S') - No backup files found in $dir" >> logs/backup_log.txt
fi

#Q20
#!/bin/bash
if [ $# -ne 1 ]; then
    echo "Usage: $0 <filename>"
    exit 1
fi
file="$1"
if [ ! -e "$file" ]; then
    echo "Error: File does not exist."
    echo "$(date '+%Y-%m-%d %H:%M:%S') - FAILED: $file does not exist" >> logs/execution_history.log
    exit 1
fi
if [ ! -r "$file" ]; then
    echo "Error: File is not readable."
    echo "$(date '+%Y-%m-%d %H:%M:%S') - FAILED: $file is not readable" >> logs/execution_history.log
    exit 1
fi
echo "File: $file"
echo "Line/Word/Character counts:"
wc "$file"
echo "$(date '+%Y-%m-%d %H:%M:%S') - SUCCESS: processed $file" >> logs/execution_history.log
