Question 2: Prime or Composite Checker

<!DOCTYPE html>
<html lang="en">
<head><title>Q2 - Prime Checker</title></head>
<body>
    <input type="number" id="numInput" placeholder="Enter number">
    <button onclick="checkPrime()">Check</button>
    <div id="output"></div>

    <script>
        function checkPrime() {
            let n = parseInt(document.getElementById("numInput").value);
            let outputDiv = document.getElementById("output");
            if (n <= 1) { outputDiv.innerHTML = "Enter number > 1"; return; }
            let isPrime = true;
            for (let i = 2; i <= Math.sqrt(n); i++) {
                if (n % i === 0) { isPrime = false; break; }
            }
            outputDiv.innerHTML = isPrime ? `${n} is Prime` : `${n} is Composite`;
        }
    </script>
</body>
</html>

Question 3: Student Grade Evaluation (if-else & switch)

<!DOCTYPE html>
<html lang="en">
<head><title>Q3 - Grade Evaluator</title></head>
<body>
    <input type="number" id="marks" placeholder="Enter marks (0-100)">
    <button onclick="calcGrade()">Get Grade</button>
    <div id="res"></div>

    <script>
        function calcGrade() {
            let m = parseFloat(document.getElementById("marks").value);
            let category = Math.floor(m / 10);
            let grade = "";
            switch (category) {
                case 10: case 9: grade = "A+"; break;
                case 8: grade = "A"; break;
                case 7: grade = "B"; break;
                case 6: grade = "C"; break;
                case 5: grade = "D"; break;
                default: grade = "F";
            }
            document.getElementById("res").innerHTML = `Grade: ${grade}`;
        }
    </script>
</body>
</html>

Question 4: Fibonacci Series Generator

<!DOCTYPE html>
<html lang="en">
<head><title>Q4 - Fibonacci</title></head>
<body>
    <input type="number" id="termCount" placeholder="Enter n terms">
    <button onclick="genFib()">Generate</button>
    <div id="out"></div>

    <script>
        function genFib() {
            let n = parseInt(document.getElementById("termCount").value);
            let fib = [];
            if (n >= 1) fib.push(0);
            if (n >= 2) fib.push(1);
            let i = 2;
            while (i < n) {
                fib.push(fib[i - 1] + fib[i - 2]);
                i++;
            }
            document.getElementById("out").innerHTML = fib.join(", ");
        }
    </script>
</body>
</html>

Question 5: String Palindrome Checker

<!DOCTYPE html>
<html lang="en">
<head><title>Q5 - Palindrome</title></head>
<body>
    <input type="text" id="strInput" placeholder="Enter string">
    <button onclick="checkPal()">Check</button>
    <div id="res"></div>

    <script>
        function checkPal() {
            let str = document.getElementById("strInput").value;
            let clean = str.toLowerCase().replace(/[^a-z0-9]/g, '');
            let rev = clean.split('').reverse().join('');
            let r = document.getElementById("res");
            r.innerHTML = (clean === rev) ? "It is a Palindrome" : "Not a Palindrome";
        }
    </script>
</body>
</html>

Question 7: Custom Object with 'this' Keyword

<!DOCTYPE html>
<html lang="en">
<head><title>Q7 - Custom Object</title></head>
<body>
    <button onclick="showObj()">Show Details</button>
    <div id="out"></div>

    <script>
        let student = {
            name: "Aman",
            marks: [80, 90, 85],
            calcAvg: function() {
                let sum = this.marks.reduce((a, b) => a + b, 0);
                return sum / this.marks.length;
            },
            getDetails: function() {
                return `Name: ${this.name}, Avg Marks: ${this.calcAvg()}`;
            }
        };
        function showObj() {
            document.getElementById("out").innerHTML = student.getDetails();
        }
    </script>
</body>

        const buttons = document.querySelectorAll(".btn");
        buttons.forEach(b => {

                if(this.innerText === "=") {

Question 20: Dynamic Table Generator

<!DOCTYPE html>
<html lang="en">
<head><title>Q20 - Table Generator</title></head>
<body>
    <input type="number" id="r" placeholder="Rows">
    <input type="number" id="c" placeholder="Cols">
    <button onclick="gen()">Generate</button>
    <div id="cont"></div>

    <script>
        function gen() {
            let rows = document.getElementById("r").value;
            let cols = document.getElementById("c").value;
            let html = "<table border='1'>";
            for(let i=0; i<rows; i++) {
                html += "<tr>";
                for(let j=0; j<cols; j++) {
                    html += `<td>Cell</td>`;
                }
                html += "</tr>";
            }
            html += "</table>";
            document.getElementById("cont").innerHTML = html;
        }
    </script>
</body>
</html>
